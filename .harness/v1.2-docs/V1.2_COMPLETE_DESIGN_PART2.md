## 3. 告警规则引擎完整设计

### 3.1 业务目标

**核心价值**:
- 基于 Prometheus 指标的智能告警
- 灵活的告警规则配置
- 多渠道告警通知
- 告警生命周期管理
- 告警分析与统计

**适用场景**:
- 主机资源告警 (CPU/内存/磁盘)
- K8s 集群告警 (Pod/Node状态)
- 应用性能告警 (响应时间/错误率)
- 业务指标告警 (自定义Metrics)

#### 3.2.3 核心服务设计

**告警评估引擎**:

```go
// backend/modules/business/observability/alert/evaluator.go

package alert

import (
    "context"
    "crypto/md5"
    "encoding/json"
    "fmt"
    "time"
    
    promapi "github.com/prometheus/client_golang/api"
    promv1 "github.com/prometheus/client_golang/api/prometheus/v1"
    "github.com/prometheus/common/model"
    "gorm.io/gorm"
)

// Evaluator evaluates alert rules
type Evaluator struct {
    db               *gorm.DB
    prometheusClient promv1.API
    interval         time.Duration
    stopCh           chan struct{}
}

// NewEvaluator creates an evaluator
func NewEvaluator(db *gorm.DB, interval time.Duration) *Evaluator {
    return &Evaluator{
        db:       db,
        interval: interval,
        stopCh:   make(chan struct{}),
    }
}

// Start begins the evaluation loop
func (e *Evaluator) Start(ctx context.Context) {
    ticker := time.NewTicker(e.interval)
    defer ticker.Stop()
    
    log.Info("alert evaluator started", "interval", e.interval)
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-e.stopCh:
            return
        case <-ticker.C:
            if err := e.evaluate(ctx); err != nil {
                log.Error("evaluation failed", "error", err)
            }
        }
    }
}

// Stop stops the evaluator
func (e *Evaluator) Stop() {
    close(e.stopCh)
}

// evaluate evaluates all enabled rules
func (e *Evaluator) evaluate(ctx context.Context) error {
    // 1. 查询所有启用的规则
    var rules []AlertRule
    if err := e.db.Where("enabled = ?", true).Find(&rules).Error; err != nil {
        return err
    }
    
    log.Debug("evaluating rules", "count", len(rules))
    
    // 2. 逐个评估规则
    for _, rule := range rules {
        if err := e.evaluateRule(ctx, &rule); err != nil {
            log.Error("rule evaluation failed", "rule", rule.RuleName, "error", err)
            
            // 记录错误
            e.db.Model(&rule).Updates(map[string]interface{}{
                "last_evaluation_at":     time.Now(),
                "last_evaluation_result": "error",
                "last_error":             err.Error(),
            })
        }
    }
    
    return nil
}

// evaluateRule evaluates a single rule
func (e *Evaluator) evaluateRule(ctx context.Context, rule *AlertRule) error {
    // 1. 获取数据源
    var datasource PrometheusDatasource
    if err := e.db.First(&datasource, rule.DatasourceID).Error; err != nil {
        return err
    }
    
    // 2. 创建 Prometheus 客户端
    promClient, err := e.createPrometheusClient(&datasource)
    if err != nil {
        return err
    }
    
    // 3. 构建 PromQL
    query, err := e.buildQuery(rule)
    if err != nil {
        return err
    }
    
    // 4. 执行查询
    result, warnings, err := promClient.Query(ctx, query, time.Now())
    if err != nil {
        return err
    }
    
    if len(warnings) > 0 {
        log.Warn("prometheus query warnings", "warnings", warnings)
    }
    
    // 5. 处理查询结果
    if err := e.processQueryResult(ctx, rule, result); err != nil {
        return err
    }
    
    // 6. 更新规则评估状态
    e.db.Model(rule).Updates(map[string]interface{}{
        "last_evaluation_at":     time.Now(),
        "last_evaluation_result": "ok",
        "last_error":             nil,
    })
    
    return nil
}

// buildQuery builds PromQL query from rule
func (e *Evaluator) buildQuery(rule *AlertRule) (string, error) {
    switch rule.RuleType {
    case "promql":
        return rule.PromQLExpr, nil
        
    case "threshold":
        // 构建阈值类型的 PromQL
        // 例: avg(node_cpu_usage{business_scope="prod"}) > 80
        query := fmt.Sprintf("%s(%s)", rule.Aggregation, rule.MetricName)
        
        if rule.TargetFilters != nil {
            var filters map[string]string
            json.Unmarshal(rule.TargetFilters, &filters)
            
            if len(filters) > 0 {
                labelSelectors := []string{}
                for k, v := range filters {
                    labelSelectors = append(labelSelectors, fmt.Sprintf(`%s="%s"`, k, v))
                }
                query = fmt.Sprintf("%s(%s{%s})", 
                    rule.Aggregation, 
                    rule.MetricName, 
                    strings.Join(labelSelectors, ","))
            }
        }
        
        return query, nil
        
    case "template":
        // 从模板构建
        var template AlertRuleTemplate
        if err := e.db.First(&template, rule.TemplateID).Error; err != nil {
            return "", err
        }
        
        // 替换模板变量
        query := template.PromQLTemplate
        if rule.TemplateParams != nil {
            var params map[string]interface{}
            json.Unmarshal(rule.TemplateParams, &params)
            
            for k, v := range params {
                query = strings.ReplaceAll(query, fmt.Sprintf("{{%s}}", k), fmt.Sprintf("%v", v))
            }
        }
        
        return query, nil
        
    default:
        return "", fmt.Errorf("unsupported rule type: %s", rule.RuleType)
    }
}

// processQueryResult processes Prometheus query result
func (e *Evaluator) processQueryResult(ctx context.Context, rule *AlertRule, result model.Value) error {
    vector, ok := result.(model.Vector)
    if !ok {
        return fmt.Errorf("expected vector result, got %T", result)
    }
    
    log.Debug("query result", "rule", rule.RuleName, "samples", len(vector))
    
    // 遍历每个时间序列
    for _, sample := range vector {
        if err := e.processSample(ctx, rule, sample); err != nil {
            log.Error("process sample failed", "error", err)
        }
    }
    
    return nil
}

// processSample processes a single sample
func (e *Evaluator) processSample(ctx context.Context, rule *AlertRule, sample *model.Sample) error {
    // 1. 计算告警指纹
    fingerprint := e.calculateFingerprint(rule, sample.Metric)
    
    // 2. 获取当前值
    currentValue := float64(sample.Value)
    
    // 3. 判断是否触发
    isFiring := e.shouldFire(rule, currentValue)
    
    // 4. 查询是否已存在活跃告警
    var existingAlert Alert
    err := e.db.Where("alert_fingerprint = ? AND status = ?", fingerprint, "firing").
        First(&existingAlert).Error
    
    if isFiring {
        if err == gorm.ErrRecordNotFound {
            // 新告警: 创建
            return e.createAlert(ctx, rule, sample, fingerprint, currentValue)
        } else {
            // 已存在: 更新
            return e.updateAlert(ctx, &existingAlert, currentValue)
        }
    } else {
        if err == nil {
            // 告警恢复
            return e.resolveAlert(ctx, &existingAlert)
        }
        // 无需处理
    }
    
    return nil
}

// shouldFire checks if alert should fire
func (e *Evaluator) shouldFire(rule *AlertRule, currentValue float64) bool {
    if rule.RuleType == "promql" {
        // PromQL 结果非空即触发
        return currentValue > 0
    }
    
    // 阈值比较
    switch rule.ThresholdOperator {
    case ">":
        return currentValue > rule.ThresholdValue
    case ">=":
        return currentValue >= rule.ThresholdValue
    case "<":
        return currentValue < rule.ThresholdValue
    case "<=":
        return currentValue <= rule.ThresholdValue
    case "==":
        return currentValue == rule.ThresholdValue
    case "!=", "<>":
        return currentValue != rule.ThresholdValue
    default:
        return false
    }
}

// calculateFingerprint calculates alert fingerprint
func (e *Evaluator) calculateFingerprint(rule *AlertRule, metric model.Metric) string {
    // 使用 rule_id + sorted labels 计算 MD5
    data := fmt.Sprintf("%d", rule.ID)
    
    // 排序标签
    labels := make([]string, 0, len(metric))
    for k, v := range metric {
        labels = append(labels, fmt.Sprintf("%s=%s", k, v))
    }
    sort.Strings(labels)
    
    data += strings.Join(labels, ",")
    
    hash := md5.Sum([]byte(data))
    return fmt.Sprintf("%x", hash)
}

// createAlert creates a new alert
func (e *Evaluator) createAlert(ctx context.Context, rule *AlertRule, sample *model.Sample, fingerprint string, currentValue float64) error {
    labels, _ := json.Marshal(sample.Metric)
    annotations, _ := json.Marshal(rule.Annotations)
    
    alert := Alert{
        RuleID:          rule.ID,
        AlertFingerprint: fingerprint,
        AlertName:       rule.RuleName,
        Status:          "firing",
        Severity:        rule.Severity,
        Labels:          labels,
        Annotations:     annotations,
        Summary:         fmt.Sprintf("%s is firing", rule.RuleName),
        CurrentValue:    currentValue,
        ThresholdValue:  rule.ThresholdValue,
        FiredAt:         time.Now(),
    }
    
    if err := e.db.Create(&alert).Error; err != nil {
        return err
    }
    
    log.Info("alert fired", "rule", rule.RuleName, "fingerprint", fingerprint)
    
    // 异步发送通知
    go e.sendNotifications(ctx, &alert, rule)
    
    return nil
}

// updateAlert updates existing alert
func (e *Evaluator) updateAlert(ctx context.Context, alert *Alert, currentValue float64) error {
    return e.db.Model(alert).Updates(map[string]interface{}{
        "current_value": currentValue,
        "updated_at":    time.Now(),
    }).Error
}

// resolveAlert resolves an alert
func (e *Evaluator) resolveAlert(ctx context.Context, alert *Alert) error {
    now := time.Now()
    duration := int(now.Sub(alert.FiredAt).Seconds())
    
    err := e.db.Model(alert).Updates(map[string]interface{}{
        "status":           "resolved",
        "resolved_at":      now,
        "duration_seconds": duration,
    }).Error
    
    if err != nil {
        return err
    }
    
    log.Info("alert resolved", "id", alert.ID, "duration", duration)
    
    // 发送恢复通知
    go e.sendRecoveryNotifications(ctx, alert)
    
    return nil
}

// sendNotifications sends alert notifications
func (e *Evaluator) sendNotifications(ctx context.Context, alert *Alert, rule *AlertRule) {
    // 1. 检查是否静默
    if e.isSilenced(alert) {
        log.Info("alert is silenced", "id", alert.ID)
        return
    }
    
    // 2. 获取通知渠道
    var channels []NotifyChannel
    if rule.NotifyChannels != nil {
        var channelNames []string
        json.Unmarshal(rule.NotifyChannels, &channelNames)
        
        e.db.Where("channel_name IN ? AND enabled = ?", channelNames, true).
            Find(&channels)
    }
    
    // 3. 发送通知
    for _, channel := range channels {
        if err := e.sendNotification(ctx, alert, rule, &channel); err != nil {
            log.Error("send notification failed", 
                "channel", channel.ChannelName, "error", err)
        }
    }
    
    // 4. 更新通知状态
    e.db.Model(alert).Updates(map[string]interface{}{
        "notified":    true,
        "notified_at": time.Now(),
    })
}

// sendNotification sends notification to a channel
func (e *Evaluator) sendNotification(ctx context.Context, alert *Alert, rule *AlertRule, channel *NotifyChannel) error {
    // 创建通知历史记录
    history := NotifyHistory{
        AlertID:   alert.ID,
        ChannelID: channel.ID,
        Subject:   fmt.Sprintf("[%s] %s", strings.ToUpper(alert.Severity), alert.AlertName),
        Content:   e.renderNotificationContent(alert, rule),
        Status:    "pending",
    }
    
    e.db.Create(&history)
    
    // 根据渠道类型发送
    var err error
    switch channel.ChannelType {
    case "email":
        err = e.sendEmail(ctx, channel, &history)
    case "slack":
        err = e.sendSlack(ctx, channel, &history)
    case "dingtalk":
        err = e.sendDingTalk(ctx, channel, &history)
    case "wecom":
        err = e.sendWeCom(ctx, channel, &history)
    case "webhook":
        err = e.sendWebhook(ctx, channel, &history)
    default:
        err = fmt.Errorf("unsupported channel type: %s", channel.ChannelType)
    }
    
    // 更新发送状态
    if err != nil {
        e.db.Model(&history).Updates(map[string]interface{}{
            "status":        "failed",
            "error_message": err.Error(),
            "attempts":      history.Attempts + 1,
        })
        return err
    }
    
    e.db.Model(&history).Updates(map[string]interface{}{
        "status":   "sent",
        "sent_at":  time.Now(),
        "attempts": history.Attempts + 1,
    })
    
    return nil
}

// isSilenced checks if alert is silenced
func (e *Evaluator) isSilenced(alert *Alert) bool {
    var count int64
    now := time.Now()
    
    e.db.Model(&Silence{}).
        Where("starts_at <= ? AND ends_at >= ?", now, now).
        Count(&count)
    
    if count == 0 {
        return false
    }
    
    // TODO: 检查标签匹配
    
    return false
}
```
