## 1. ArgoCD GitOps 完整设计

### 1.1 业务目标

**核心价值**: 
- 实现声明式、自动化的 Kubernetes 应用交付
- Git 作为唯一事实源 (Single Source of Truth)
- 自动同步、自动回滚、完整审计

**适用场景**:
- 微服务应用的持续交付
- 多环境 (dev/test/prod) 配置管理
- GitOps 最佳实践落地

### 1.2 架构设计

package argocd

import (
    "context"
    "crypto/tls"
    "encoding/json"

    "io"
    "net/http"
    "time"
)

// Client wraps ArgoCD API operations
type Client struct {
    serverURL  string
    token      string
    httpClient *http.Client
}

// NewClient creates an ArgoCD API client
func NewClient(serverURL, token string, insecureSkipVerify bool) *Client {
    tr := &http.Transport{
        TLSClientConfig: &tls.Config{InsecureSkipVerify: insecureSkipVerify},
    }
    
    return &Client{
        serverURL: serverURL,
        token:     token,
        httpClient: &http.Client{
            Timeout:   30 * time.Second,
            Transport: tr,
        },
    }
}

// HealthCheck tests connectivity to ArgoCD server
func (c *Client) HealthCheck(ctx context.Context) error {
    req, err := http.NewRequestWithContext(ctx, "GET", 
        fmt.Sprintf("%s/api/version", c.serverURL), nil)
    if err != nil {
        return err
    }
    
    req.Header.Set("Authorization", fmt.Sprintf("Bearer %s", c.token))
    
    resp, err := c.httpClient.Do(req)
    if err != nil {
        return fmt.Errorf("connection failed: %w", err)
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return fmt.Errorf("unexpected status: %d", resp.StatusCode)
    }
    
    return nil
}

// Application represents ArgoCD Application
type Application struct {
    Metadata struct {
        Name      string `json:"name"`
        Namespace string `json:"namespace"`
    } `json:"metadata"`
    Spec struct {
        Source struct {
            RepoURL        string `json:"repoURL"`
            Path           string `json:"path"`
            TargetRevision string `json:"targetRevision"`
        } `json:"source"`
        Destination struct {
            Server    string `json:"server"`
            Namespace string `json:"namespace"`
        } `json:"destination"`
        SyncPolicy *struct {
            Automated *struct {
                Prune    bool `json:"prune"`
                SelfHeal bool `json:"selfHeal"`
            } `json:"automated,omitempty"`
        } `json:"syncPolicy,omitempty"`
    } `json:"spec"`
    Status struct {
        Sync struct {
            Status   string `json:"status"` // Synced/OutOfSync
            Revision string `json:"revision"`
        } `json:"sync"`
        Health struct {
            Status  string `json:"status"` // Healthy/Progressing/Degraded
            Message string `json:"message,omitempty"`
        } `json:"health"`
        OperationState *struct {
            Phase     string    `json:"phase"` // Running/Succeeded/Failed
            Message   string    `json:"message,omitempty"`
            StartedAt time.Time `json:"startedAt"`
            FinishedAt *time.Time `json:"finishedAt,omitempty"`
        } `json:"operationState,omitempty"`
        Resources []struct {
            Kind      string `json:"kind"`
            Name      string `json:"name"`
            Namespace string `json:"namespace"`
            Status    string `json:"status"` // Synced/OutOfSync
            Health    struct {
                Status  string `json:"status"`
                Message string `json:"message,omitempty"`
            } `json:"health"`
        } `json:"resources"`
    } `json:"status"`
}

// CreateApplication creates a new ArgoCD Application
func (c *Client) CreateApplication(ctx context.Context, app *Application) (*Application, error) {
    body, _ := json.Marshal(app)
    
    req, err := http.NewRequestWithContext(ctx, "POST",
        fmt.Sprintf("%s/api/v1/applications", c.serverURL), 
        bytes.NewReader(body))
    if err != nil {
        return nil, err
    }
    
    req.Header.Set("Authorization", fmt.Sprintf("Bearer %s", c.token))
    req.Header.Set("Content-Type", "application/json")
    
    resp, err := c.httpClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK && resp.StatusCode != http.StatusCreated {
        bodyBytes, _ := io.ReadAll(resp.Body)
        return nil, fmt.Errorf("create failed: %d %s", resp.StatusCode, string(bodyBytes))
    }
    
    var result Application
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }
    
    return &result, nil
}

    }
    
    return &app, nil
}

// backend/modules/business/cicd/argocd/reconciler.go

package argocd

import (
    "context"

// Reconciler synchronizes ArgoCD Application state
type Reconciler struct {
    db       *gorm.DB
    interval time.Duration
    stopCh   chan struct{}
}

// NewReconciler creates a reconciler
func NewReconciler(db *gorm.DB, interval time.Duration) *Reconciler {
    return &Reconciler{
        db:       db,
        interval: interval,
        stopCh:   make(chan struct{}),
    }
}

// Start begins the reconciliation loop
func (r *Reconciler) Start(ctx context.Context) {
    ticker := time.NewTicker(r.interval)
    defer ticker.Stop()
    
    log.Info("ArgoCD reconciler started", "interval", r.interval)
    
    for {
        select {
        case <-ctx.Done():
            log.Info("ArgoCD reconciler stopped")
            return
        case <-r.stopCh:
            log.Info("ArgoCD reconciler stopped")
            return
        case <-ticker.C:
            if err := r.reconcile(ctx); err != nil {
                log.Error("reconciliation failed", "error", err)
            }
        }
    }
}

// Stop stops the reconciler
func (r *Reconciler) Stop() {
    close(r.stopCh)
}

// reconcile synchronizes all applications
func (r *Reconciler) reconcile(ctx context.Context) error {
    // 1. 查询所有 ArgoCD Server
    var servers []ArgoCDServer
    if err := r.db.Where("status = ?", "active").Find(&servers).Error; err != nil {
        return err
    }
    
    for _, server := range servers {
        if err := r.reconcileServer(ctx, &server); err != nil {
            log.Error("server reconciliation failed", 
                "server", server.Name, "error", err)
            
            // 更新服务器状态
            r.db.Model(&server).Updates(map[string]interface{}{
                "status":           "unreachable",
                "last_check_at":    time.Now(),
                "last_check_error": err.Error(),
            })
        } else {
            // 更新服务器状态为正常
            r.db.Model(&server).Updates(map[string]interface{}{
                "status":           "active",
                "last_check_at":    time.Now(),
                "last_check_error": nil,
            })
        }
    }
    
    return nil
}

// reconcileServer synchronizes applications for one server
func (r *Reconciler) reconcileServer(ctx context.Context, server *ArgoCDServer) error {
    // 1. 解密 token
    token, err := decrypt(server.APITokenEncrypted)
    if err != nil {
        return fmt.Errorf("decrypt token failed: %w", err)
    }
    
    // 2. 创建 ArgoCD client
    client := NewClient(server.ServerURL, token, server.InsecureSkipVerify)
    
    // 3. 健康检查
    if err := client.HealthCheck(ctx); err != nil {
        return err
    }
    
    // 4. 查询该服务器的所有 Application
    var apps []ArgoCDApplication
    if err := r.db.Where("argocd_server_id = ?", server.ID).Find(&apps).Error; err != nil {
        return err
    }
    
    // 5. 同步每个 Application 状态
    for _, app := range apps {
        if err := r.reconcileApplication(ctx, client, &app); err != nil {
            log.Error("application reconciliation failed",
                "app", app.ArgoCDAppName, "error", err)
        }
    }
    
    return nil
}

// reconcileApplication synchronizes one application
func (r *Reconciler) reconcileApplication(ctx context.Context, client *Client, app *ArgoCDApplication) error {
    // 1. 从 ArgoCD 获取最新状态
    argoApp, err := client.GetApplication(ctx, app.ArgoCDAppName)
    if err != nil {
        return err
    }
    
    // 2. 检测状态变化
    stateChanged := app.SyncStatus != argoApp.Status.Sync.Status ||
                    app.HealthStatus != argoApp.Status.Health.Status ||
                    app.CurrentRevision != argoApp.Status.Sync.Revision
    
    // 3. 更新本地 Application 状态
    updates := map[string]interface{}{
        "sync_status":     argoApp.Status.Sync.Status,
        "health_status":   argoApp.Status.Health.Status,
        "current_revision": argoApp.Status.Sync.Revision,
        "last_synced_at":  time.Now(),
    }
    
    if argoApp.Status.OperationState != nil {
        updates["last_operation_state"] = argoApp.Status.OperationState.Phase
        updates["last_operation_message"] = argoApp.Status.OperationState.Message
    }
    
    if err := r.db.Model(app).Updates(updates).Error; err != nil {
        return err
    }
    
    // 4. 更新资源清单
    if err := r.updateAppResources(app.ID, argoApp.Status.Resources); err != nil {
        log.Error("update resources failed", "error", err)
    }
    
    // 5. 如果状态变化，更新关联的服务实例
    if stateChanged && app.ServiceInstanceID != nil {
        if err := r.updateServiceInstance(ctx, app); err != nil {
            log.Error("update service instance failed", "error", err)
        }
    }
    
    // 6. 记录同步历史（如果有新的操作）
    if argoApp.Status.OperationState != nil {
        if err := r.recordSyncHistory(app, argoApp.Status.OperationState); err != nil {
            log.Error("record sync history failed", "error", err)
        }
    }
    
    return nil
}

// updateAppResources updates resource manifest
func (r *Reconciler) updateAppResources(appID uint64, resources []struct{...}) error {
    // 删除旧资源
    r.db.Where("application_id = ?", appID).Delete(&ArgoCDAppResource{})
    
    // 插入新资源
    for _, res := range resources {
        resource := ArgoCDAppResource{
            ApplicationID:    appID,
            ResourceKind:     res.Kind,
            ResourceName:     res.Name,
            ResourceNamespace: res.Namespace,
            SyncStatus:       res.Status,
            HealthStatus:     res.Health.Status,
            HealthMessage:    res.Health.Message,
        }
        
        if err := r.db.Create(&resource).Error; err != nil {
            return err
        }
    }
    
    return nil
}

// updateServiceInstance updates associated service instance state
func (r *Reconciler) updateServiceInstance(ctx context.Context, app *ArgoCDApplication) error {
    // 调用服务实例状态更新 capability
    // (需要注入 ServiceInstanceStateCommand)
    
    observedState := "unknown"
    healthState := "unknown"
    
    switch app.SyncStatus {
    case "Synced":
        observedState = "deployed"
    case "OutOfSync":
        observedState = "drift"
    }
    
    switch app.HealthStatus {
    case "Healthy":
        healthState = "healthy"
    case "Progressing":
        healthState = "progressing"
    case "Degraded":
        healthState = "degraded"
    }
    
    // 实际调用需要通过 capability 接口
    log.Info("service instance state updated",
        "instance_id", *app.ServiceInstanceID,
        "observed_state", observedState,
        "health_state", healthState)
    
    return nil
}

// recordSyncHistory records sync operation history
func (r *Reconciler) recordSyncHistory(app *ArgoCDApplication, opState *OperationState) error {
    // 检查是否已记录
    var count int64
    r.db.Model(&ArgoCDSyncHistory{}).
        Where("application_id = ? AND started_at = ?", app.ID, opState.StartedAt).
        Count(&count)
    
    if count > 0 {
        return nil // 已记录，跳过
    }
    
    history := ArgoCDSyncHistory{
        ApplicationID:    app.ID,
        OperationType:    "sync", // 需要从 opState 推断
        InitiatedBy:      "auto",
        Revision:         app.CurrentRevision,
        Phase:            opState.Phase,
        Message:          opState.Message,
        StartedAt:        opState.StartedAt,
        FinishedAt:       opState.FinishedAt,
    }
    
    if opState.FinishedAt != nil {
        duration := opState.FinishedAt.Sub(opState.StartedAt)
        history.DurationSeconds = int(duration.Seconds())
    }
    
    return r.db.Create(&history).Error
}

// backend/modules/business/k8s/backup/velero_client.go

package backup

import (
    "context"
    "fmt"
    
    velerov1 "github.com/vmware-tanzu/velero/pkg/apis/velero/v1"
    metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
    "k8s.io/client-go/kubernetes"
    "k8s.io/client-go/rest"
)

// VeleroClient wraps Velero CRD operations
type VeleroClient struct {
    k8sClient     *kubernetes.Clientset
    veleroClient  veleroclientset.Interface
    namespace     string
}

// NewVeleroClient creates a Velero client
func NewVeleroClient(kubeconfig *rest.Config, namespace string) (*VeleroClient, error) {
    k8sClient, err := kubernetes.NewForConfig(kubeconfig)
    if err != nil {
        return nil, err
    }
    
    veleroClient, err := veleroclientset.NewForConfig(kubeconfig)
    if err != nil {
        return nil, err
    }
    
    return &VeleroClient{
        k8sClient:    k8sClient,
        veleroClient: veleroClient,
        namespace:    namespace,
    }, nil
}

// CreateBackup creates a Velero Backup CRD
func (c *VeleroClient) CreateBackup(ctx context.Context, backup *velerov1.Backup) (*velerov1.Backup, error) {
    return c.veleroClient.VeleroV1().Backups(c.namespace).Create(ctx, backup, metav1.CreateOptions{})
}

// GetBackup retrieves a Backup
func (c *VeleroClient) GetBackup(ctx context.Context, name string) (*velerov1.Backup, error) {
    return c.veleroClient.VeleroV1().Backups(c.namespace).Get(ctx, name, metav1.GetOptions{})
}

// ListBackups lists all Backups
func (c *VeleroClient) ListBackups(ctx context.Context) (*velerov1.BackupList, error) {
    return c.veleroClient.VeleroV1().Backups(c.namespace).List(ctx, metav1.ListOptions{})
}

// DeleteBackup deletes a Backup
func (c *VeleroClient) DeleteBackup(ctx context.Context, name string) error {
    return c.veleroClient.VeleroV1().Backups(c.namespace).Delete(ctx, name, metav1.DeleteOptions{})
}

// CreateSchedule creates a Velero Schedule CRD

    return c.veleroClient.VeleroV1().Schedules(c.namespace).Create(ctx, schedule, metav1.CreateOptions{})

// GetBackupLogs retrieves backup logs
func (c *VeleroClient) GetBackupLogs(ctx context.Context, backupName string) (string, error) {
    // 从 velero pod 获取日志
    pods, err := c.k8sClient.CoreV1().Pods(c.namespace).List(ctx, metav1.ListOptions{
        LabelSelector: "component=velero",
    })
    if err != nil {
        return "", err
    }
    
    if len(pods.Items) == 0 {
        return "", fmt.Errorf("velero pod not found")
    }
    
    podName := pods.Items[0].Name
    
    req := c.k8sClient.CoreV1().Pods(c.namespace).GetLogs(podName, &corev1.PodLogOptions{
        Container: "velero",
    })
    
    logs, err := req.Stream(ctx)
    if err != nil {
        return "", err
    }
    defer logs.Close()
    
    buf, err := io.ReadAll(logs)
    if err != nil {
        return "", err
    }
    
    return string(buf), nil
}
