```go
// backend/modules/business/filedist/transfer_service.go

package filedist

import (
    "context"
    "crypto/sha256"
    "fmt"
    "io"
    "os"
    "path/filepath"
    "time"
    
    "github.com/pkg/sftp"
    "golang.org/x/crypto/ssh"
    "gorm.io/gorm"
)

// TransferService handles file distribution
type TransferService struct {
    db              *gorm.DB
    fileStorageRoot string
    maxConcurrency  int
    taskQueue       chan *TaskHost
    stopCh          chan struct{}
}

// NewTransferService creates a transfer service
func NewTransferService(db *gorm.DB, storageRoot string, maxConcurrency int) *TransferService {
    return &TransferService{
        db:              db,
        fileStorageRoot: storageRoot,
        maxConcurrency:  maxConcurrency,
        taskQueue:       make(chan *TaskHost, 1000),
        stopCh:          make(chan struct{}),
    }
}

// Start starts worker pool
func (s *TransferService) Start(ctx context.Context) {
    log.Info("file distribution service started", "workers", s.maxConcurrency)
    
    // 启动 Worker 池
    for i := 0; i < s.maxConcurrency; i++ {
        go s.worker(ctx, i)
    }
    
    // 启动任务调度器
    go s.scheduler(ctx)
}

// Stop stops the service
func (s *TransferService) Stop() {
    close(s.stopCh)
}

// scheduler schedules pending tasks
func (s *TransferService) scheduler(ctx context.Context) {
    ticker := time.NewTicker(5 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-s.stopCh:
            return
        case <-ticker.C:
            s.schedulePendingTasks(ctx)
        }
    }
}

// schedulePendingTasks schedules pending task hosts
func (s *TransferService) schedulePendingTasks(ctx context.Context) {
    // 查询 pending 状态的任务主机
    var taskHosts []TaskHost
    err := s.db.Where("status = ?", "pending").
        Order("created_at ASC").
        Limit(100).
        Find(&taskHosts).Error
    
    if err != nil {
        log.Error("query pending tasks failed", "error", err)
        return
    }
    
    // 加入队列
    for i := range taskHosts {
        select {
        case s.taskQueue <- &taskHosts[i]:
            // 标记为 queued
            s.db.Model(&taskHosts[i]).Update("status", "queued")
        default:
            // 队列已满，下次再试
            return
        }
    }
}

// worker processes file distribution tasks
func (s *TransferService) worker(ctx context.Context, workerID int) {
    log.Info("file distribution worker started", "id", workerID)
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-s.stopCh:
            return
        case taskHost := <-s.taskQueue:
            s.processTaskHost(ctx, workerID, taskHost)
        }
    }
}

// processTaskHost processes a single task host
func (s *TransferService) processTaskHost(ctx context.Context, workerID int, taskHost *TaskHost) {
    log.Info("processing file distribution", 
        "worker", workerID, 
        "task", taskHost.TaskID, 
        "host", taskHost.HostIP)
    
    // 更新状态为 running
    s.db.Model(taskHost).Updates(map[string]interface{}{
        "status":     "running",
        "started_at": time.Now(),
        "worker_id":  fmt.Sprintf("worker-%d", workerID),
    })
    
    // 记录事件
    s.recordEvent(taskHost.TaskID, taskHost.HostID, "host_start", 
        fmt.Sprintf("开始分发到主机 %s", taskHost.HostIP), nil)
    
    // 执行分发
    err := s.distributeFile(ctx, taskHost)
    
    if err != nil {
        // 分发失败
        s.db.Model(taskHost).Updates(map[string]interface{}{
            "status":        "failed",
            "finished_at":   time.Now(),
            "error_message": err.Error(),
        })
        
        s.recordEvent(taskHost.TaskID, taskHost.HostID, "host_finish",
            fmt.Sprintf("分发失败: %s", err.Error()), nil)
        
        // 更新任务统计
        s.updateTaskStats(taskHost.TaskID)
        
        log.Error("file distribution failed", 
            "task", taskHost.TaskID, 
            "host", taskHost.HostIP, 
            "error", err)
        return
    }
    
    // 分发成功
    duration := time.Since(taskHost.StartedAt).Seconds()
    s.db.Model(taskHost).Updates(map[string]interface{}{
        "status":           "success",
        "finished_at":      time.Now(),
        "duration_seconds": int(duration),
    })
    
    s.recordEvent(taskHost.TaskID, taskHost.HostID, "host_finish",
        fmt.Sprintf("分发成功，耗时 %.2f 秒", duration), nil)
    
    // 更新任务统计
    s.updateTaskStats(taskHost.TaskID)
    
    log.Info("file distribution succeeded",
        "task", taskHost.TaskID,
        "host", taskHost.HostIP,
        "duration", duration)
}

// distributeFile distributes file to a host
func (s *TransferService) distributeFile(ctx context.Context, taskHost *TaskHost) error {
    // 1. 获取任务详情
    var task Task
    if err := s.db.First(&task, taskHost.TaskID).Error; err != nil {
        return fmt.Errorf("get task failed: %w", err)
    }
    
    // 2. 获取文件详情
    var file File
    if err := s.db.First(&file, task.FileID).Error; err != nil {
        return fmt.Errorf("get file failed: %w", err)
    }
    
    // 3. 获取主机详情 (SSH连接信息)
    host, err := s.getHostInfo(taskHost.HostID)
    if err != nil {
        return fmt.Errorf("get host info failed: %w", err)
    }
    
    // 4. 建立 SSH 连接
    sshClient, err := s.connectSSH(host)
    if err != nil {
        return fmt.Errorf("ssh connect failed: %w", err)
    }
    defer sshClient.Close()
    
    // 5. 建立 SFTP 会话
    sftpClient, err := sftp.NewClient(sshClient)
    if err != nil {
        return fmt.Errorf("sftp session failed: %w", err)
    }
    defer sftpClient.Close()
    
    // 6. 检查目标路径
    targetDir := filepath.Dir(task.TargetPath)
    if task.CreateParentDirs {
        if err := sftpClient.MkdirAll(targetDir); err != nil {
            return fmt.Errorf("create parent dirs failed: %w", err)
        }
    }
    
    // 7. 检查文件是否已存在
    if !task.ForceOverwrite {
        if _, err := sftpClient.Stat(task.TargetPath); err == nil {
            if task.BackupExisting {
                // 备份已存在文件
                backupPath := fmt.Sprintf("%s.bak.%d", task.TargetPath, time.Now().Unix())
                if err := sftpClient.Rename(task.TargetPath, backupPath); err != nil {
                    return fmt.Errorf("backup existing file failed: %w", err)
                }
            } else {
                return fmt.Errorf("file already exists: %s", task.TargetPath)
            }
        }
    }
    
    // 8. 传输文件
    if err := s.transferFile(ctx, sftpClient, file, task, taskHost); err != nil {
        return fmt.Errorf("transfer failed: %w", err)
    }
    
    // 9. 校验完整性
    if task.VerifyChecksum {
        if err := s.verifyChecksum(ctx, sshClient, sftpClient, file, task, taskHost); err != nil {
            return fmt.Errorf("checksum verification failed: %w", err)
        }
    }
    
    // 10. 设置权限
    if err := s.setPermissions(sftpClient, sshClient, task); err != nil {
        return fmt.Errorf("set permissions failed: %w", err)
    }
    
    return nil
}

// transferFile transfers file via SFTP
func (s *TransferService) transferFile(ctx context.Context, sftpClient *sftp.Client, file *File, task *Task, taskHost *TaskHost) error {
    // 打开本地文件
    localPath := filepath.Join(s.fileStorageRoot, file.FilePath)
    srcFile, err := os.Open(localPath)
    if err != nil {
        return fmt.Errorf("open local file failed: %w", err)
    }
    defer srcFile.Close()
    
    // 创建远程文件
    dstFile, err := sftpClient.Create(task.TargetPath)
    if err != nil {
        return fmt.Errorf("create remote file failed: %w", err)
    }
    defer dstFile.Close()
    
    // 分块传输 + 进度追踪
    buffer := make([]byte, 1024*1024) // 1MB buffer
    var transferred int64 = 0
    totalBytes := file.FileSize
    
    // 更新初始进度
    s.db.Model(taskHost).Updates(map[string]interface{}{
        "total_bytes": totalBytes,
        "bytes_transferred": 0,
        "progress_percent": 0.00,
    })
    
    lastProgressUpdate := time.Now()
    
    for {
        // 检查取消
        select {
        case <-ctx.Done():
            return fmt.Errorf("transfer canceled")
        default:
        }
        
        // 读取块
        n, err := srcFile.Read(buffer)
        if n > 0 {
            // 写入远程
            _, writeErr := dstFile.Write(buffer[:n])
            if writeErr != nil {
                return fmt.Errorf("write remote file failed: %w", writeErr)
            }
            
            transferred += int64(n)
            
            // 更新进度 (每秒更新一次)
            if time.Since(lastProgressUpdate) >= time.Second {
                progress := float64(transferred) / float64(totalBytes) * 100
                speed := float64(transferred) / time.Since(taskHost.StartedAt).Seconds() / 1024 / 1024 * 8 // Mbps
                
                s.db.Model(taskHost).Updates(map[string]interface{}{
                    "bytes_transferred":   transferred,
                    "progress_percent":    progress,
                    "transfer_speed_mbps": speed,
                })
                
                s.recordEvent(task.ID, taskHost.HostID, "transfer_progress",
                    fmt.Sprintf("传输进度 %.2f%%", progress),
                    map[string]interface{}{
                        "bytes_transferred": transferred,
                        "total_bytes":       totalBytes,
                        "speed_mbps":        speed,
                    })
                
                lastProgressUpdate = time.Now()
            }
        }
        
        if err == io.EOF {
            break
        }
        
        if err != nil {
            return fmt.Errorf("read local file failed: %w", err)
        }
    }
    
    // 最终进度更新
    s.db.Model(taskHost).Updates(map[string]interface{}{
        "bytes_transferred": transferred,
        "progress_percent":  100.00,
    })
    
    return nil
}

// verifyChecksum verifies file checksum on remote host
func (s *TransferService) verifyChecksum(ctx context.Context, sshClient *ssh.Client, sftpClient *sftp.Client, file *File, task *Task, taskHost *TaskHost) error {
    s.recordEvent(task.ID, taskHost.HostID, "checksum", "开始校验文件完整性", nil)
    
    // 在远程主机计算 SHA-256
    session, err := sshClient.NewSession()
    if err != nil {
        return fmt.Errorf("create ssh session failed: %w", err)
    }
    defer session.Close()
    
    cmd := fmt.Sprintf("sha256sum %s | awk '{print $1}'", task.TargetPath)
    output, err := session.CombinedOutput(cmd)
    if err != nil {
        return fmt.Errorf("calculate remote checksum failed: %w", err)
    }
    
    remoteSHA256 := strings.TrimSpace(string(output))
    
    // 比对哈希
    if remoteSHA256 != file.SHA256 {
        s.recordEvent(task.ID, taskHost.HostID, "checksum",
            fmt.Sprintf("校验失败: 期望 %s, 实际 %s", file.SHA256, remoteSHA256), nil)
        
        return fmt.Errorf("checksum mismatch: expected %s, got %s", file.SHA256, remoteSHA256)
    }
    
    // 更新校验结果
    s.db.Model(taskHost).Updates(map[string]interface{}{
        "remote_sha256":     remoteSHA256,
        "checksum_verified": true,
    })
    
    s.recordEvent(task.ID, taskHost.HostID, "checksum", "校验通过", nil)
    
    return nil
}

// setPermissions sets file permissions
func (s *TransferService) setPermissions(sftpClient *sftp.Client, sshClient *ssh.Client, task *Task) error {
    // 1. chmod
    if task.TargetMode != "" {
        mode, err := strconv.ParseUint(task.TargetMode, 8, 32)
        if err == nil {
            if err := sftpClient.Chmod(task.TargetPath, os.FileMode(mode)); err != nil {
                return fmt.Errorf("chmod failed: %w", err)
            }
        }
    }
    
    // 2. chown (需要通过 SSH 执行)
    if task.TargetOwner != "" || task.TargetGroup != "" {
        session, err := sshClient.NewSession()
        if err != nil {
            return fmt.Errorf("create ssh session failed: %w", err)
        }
        defer session.Close()
        
        owner := task.TargetOwner
        if owner == "" {
            owner = "root"
        }
        
        group := task.TargetGroup
        if group == "" {
            group = owner
        }
        
        cmd := fmt.Sprintf("chown %s:%s %s", owner, group, task.TargetPath)
        if err := session.Run(cmd); err != nil {
            return fmt.Errorf("chown failed: %w", err)
        }
    }
    
    return nil
}

// updateTaskStats updates task statistics
func (s *TransferService) updateTaskStats(taskID uint64) {
    var stats struct {
        Total   int
        Success int
        Failed  int
        Running int
    }
    
    s.db.Model(&TaskHost{}).Where("task_id = ?", taskID).
        Select("COUNT(*) as total, "+
            "SUM(CASE WHEN status = 'success' THEN 1 ELSE 0 END) as success, "+
            "SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END) as failed, "+
            "SUM(CASE WHEN status IN ('running', 'queued') THEN 1 ELSE 0 END) as running").
        Scan(&stats)
    
    updates := map[string]interface{}{
        "success_hosts": stats.Success,
        "failed_hosts":  stats.Failed,
        "running_hosts": stats.Running,
    }
    
    // 判断任务是否完成
    if stats.Running == 0 {
        if stats.Failed > 0 {
            updates["status"] = "failed"
        } else {
            updates["status"] = "success"
        }
        updates["finished_at"] = time.Now()
        
        s.recordEvent(taskID, 0, "task_finish",
            fmt.Sprintf("任务完成: 成功 %d, 失败 %d", stats.Success, stats.Failed), nil)
    }
    
    s.db.Model(&Task{}).Where("id = ?", taskID).Updates(updates)
}

// recordEvent records an event
func (s *TransferService) recordEvent(taskID uint64, hostID uint64, eventType, message string, details map[string]interface{}) {
    detailsJSON, _ := json.Marshal(details)
    
    event := Event{
        TaskID:     taskID,
        HostID:     hostID,
        EventType:  eventType,
        Message:    message,
        Details:    detailsJSON,
        OccurredAt: time.Now(),
    }
    
    s.db.Create(&event)
}
```

```go
// backend/modules/business/service/health_checker.go

type HealthChecker struct {
    db       *gorm.DB
    interval time.Duration
}

func (c *HealthChecker) Start(ctx context.Context) {
    ticker := time.NewTicker(c.interval)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            c.checkAllInstances(ctx)
        }
    }
}

func (c *HealthChecker) checkAllInstances(ctx context.Context) {
    var instances []ServiceInstance
    c.db.Where("desired_state = ?", "running").Find(&instances)
    
    for _, instance := range instances {
        c.checkInstance(ctx, &instance)
    }
}

func (c *HealthChecker) checkInstance(ctx context.Context, instance *ServiceInstance) {
    switch instance.TargetType {
    case "vm":
        c.checkVMInstance(ctx, instance)
    case "k8s":
        c.checkK8sInstance(ctx, instance)
    }
}

func (c *HealthChecker) checkVMInstance(ctx context.Context, instance *ServiceInstance) {
    // 解析 target_ref
    var target struct {
        HostID     uint64 `json:"hostId"`
        Port       int    `json:"port"`
        HealthPath string `json:"healthPath"`
    }
    json.Unmarshal(instance.TargetRef, &target)
    
    // 获取主机信息
    host, err := c.getHost(target.HostID)
    if err != nil {
        c.updateHealth(instance, "unknown", err.Error())
        return
    }
    
    // HTTP 健康检查
    healthURL := fmt.Sprintf("http://%s:%d%s", host.IP, target.Port, target.HealthPath)
    resp, err := http.Get(healthURL)
    if err != nil {
        c.updateHealth(instance, "unhealthy", err.Error())
        return
    }
    defer resp.Body.Close()
    
    if resp.StatusCode == 200 {
        c.updateHealth(instance, "healthy", "")
    } else {
        c.updateHealth(instance, "unhealthy", fmt.Sprintf("HTTP %d", resp.StatusCode))
    }
}

func (c *HealthChecker) checkK8sInstance(ctx context.Context, instance *ServiceInstance) {
    // 解析 k8s target
    var target struct {
        ClusterID    uint64 `json:"clusterId"`
        Namespace    string `json:"namespace"`
        WorkloadKind string `json:"workloadKind"`
        WorkloadName string `json:"workloadName"`
    }
    json.Unmarshal(instance.TargetRef, &target)
    
    // 获取 K8s 客户端
    k8sClient, err := c.getK8sClient(target.ClusterID)
    if err != nil {
        c.updateHealth(instance, "unknown", err.Error())
        return
    }
    
    // 查询 Workload 状态
    health, message := c.getWorkloadHealth(k8sClient, target)
    c.updateHealth(instance, health, message)
}

func (c *HealthChecker) updateHealth(instance *ServiceInstance, health, message string) {
    c.db.Model(instance).Updates(map[string]interface{}{
        "health_state":   health,
        "health_message": message,
        "health_revision": time.Now().Unix(),
    })
}
```

```go
// backend/pkg/encryption/encryption.go

// 使用 AES-256-GCM 加密
// 密钥从环境变量或配置文件读取
// 格式: v1:base64(nonce+ciphertext+tag)

func Encrypt(plaintext string) (string, error) {
    key := getEncryptionKey() // 32 bytes
    
    block, err := aes.NewCipher(key)
    if err != nil {
        return "", err
    }
    
    gcm, err := cipher.NewGCM(block)
    if err != nil {
        return "", err
    }
    
    nonce := make([]byte, gcm.NonceSize())
    if _, err := io.ReadFull(rand.Reader, nonce); err != nil {
        return "", err
    }
    
    ciphertext := gcm.Seal(nonce, nonce, []byte(plaintext), nil)
    encoded := base64.StdEncoding.EncodeToString(ciphertext)
    
    return "v1:" + encoded, nil
}

func Decrypt(encrypted string) (string, error) {
    if !strings.HasPrefix(encrypted, "v1:") {
        return "", fmt.Errorf("invalid encryption format")
    }
    
    encoded := strings.TrimPrefix(encrypted, "v1:")
    ciphertext, err := base64.StdEncoding.DecodeString(encoded)
    if err != nil {
        return "", err
    }
    
    key := getEncryptionKey()
    block, err := aes.NewCipher(key)
    if err != nil {
        return "", err
    }
    
    gcm, err := cipher.NewGCM(block)
    if err != nil {
        return "", err
    }
    
    nonceSize := gcm.NonceSize()
    if len(ciphertext) < nonceSize {
        return "", fmt.Errorf("ciphertext too short")
    }
    
    nonce, ciphertext := ciphertext[:nonceSize], ciphertext[nonceSize:]
    plaintext, err := gcm.Open(nil, nonce, ciphertext, nil)
    if err != nil {
        return "", err
    }
    
    return string(plaintext), nil
}
```
